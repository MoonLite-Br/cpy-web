Fibonacci (fib)

Calculates the nth Fibonacci number recursively.

fn fib(n) {
    if (n <= 1) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}

print(fib(10)); // 55

Point Structure

Defines a 2D Point type with vector addition.

struct Point {
    x: int,
    y: int
}

fn add_points(p1: Point, p2: Point) -> Point {
    return Point {
        x: p1.x + p2.x,
        y: p1.y + p2.y
    };
}

Counter Object

Simple stateful counter class pattern.

class Counter {
    private count: int = 0;

    public fn increment() {
        self.count += 1;
    }

    public fn get_value() -> int {
        return self.count;
    }
}
