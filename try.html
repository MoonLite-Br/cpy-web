# fib-ish generator
def fib():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

print([n for _, n in zip(range(10), fib())])

# classes + f-strings
class Point:
    def __init__(self, x, y):
        self.x, self.y = x, y
    def __repr__(self):
        return f"Point({self.x}, {self.y})"

print(Point(2, 3))

# collections
from collections import Counter
print(Counter("mississippi").most_common(2))
